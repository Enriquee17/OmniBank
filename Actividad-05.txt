Actividad 5: Levantando los Cimientos del Core Bancario OmniBank (DDL)

1. El espacio modular
CREATE SCHEMA IF NOT EXISTS core;

2. Construyendo la entidad raíz

CREATE TABLE core.customers (
   cliente_id UUID DEFAULT gen_random_uuid(),
   nombre_completo VARCHAR(100) NOT NULL,
   correo VARCHAR(100) NOT NULL,
   teléfono VARCHAR(100) NOT NULL,
   tax_id VARCHAR(100) NOT NULL,
   created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
   is_active BOOLEAN DEFAULT true NOT NULL,

   CONSTRAINT pk_clientes PRIMARY KEY (cliente_id),
   CONSTRAINT uq_clientes_correo UNIQUE (correo),
   CONSTRAINT uq_clientes_tax_id UNIQUE (tax_id)
   );

3. Construyendo las entidades hijas

CREATE TABLE core.accounts (
   cuenta_id UUID DEFAULT gen_random_uuid(),
   cliente_id UUID NOT NULL,
   tipo tipo_cuenta DEAFULT 'CHECKING' NOT NULL,
   saldo monto_financiero NOT NULL,
   moneda moneda DEFAULT 'USD' NOT NULL,

   CONSTRAINT pk_cuentas PRIMARY KEY (cuenta_id),
   CONSTRAINT fk_cuenta_cliente FOREIGN KEY (cliente_id)
   REFERENCES core.customers (cliente_id) ON DELETE RESTRICT
   );
