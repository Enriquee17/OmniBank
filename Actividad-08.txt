Actividad 8: Extracción de Reportes Gerenciales en OmniBank y Detección de Vacíos

Reporte 1

SELECT account_number, account_type, balance
FROM core.accounts
WHERE balance >= 100.00 AND account_type IN ('SAVINGS')
ORDER BY balance DESC
LIMIT 2;

Reporte 2

SELECT first_name, last_name, email
FROM core.customers
WHERE email ILIKE '%@email.com';

Reporte 3

SELECT account_id, tx_type, amount,
COALESCE(description,'COBRO INSTITUCIONAL O COMISION - SIN RECEPTOR') AS descripcion_visible
FROM core.transactions
WHERE description IS NULL;
